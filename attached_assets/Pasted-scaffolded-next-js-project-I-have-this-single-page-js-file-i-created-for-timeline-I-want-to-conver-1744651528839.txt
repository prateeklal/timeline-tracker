        let modalTitle = null;
        let modalBody = null;

        // Wait for the DOM to be fully loaded before getting elements
        document.addEventListener('DOMContentLoaded', (event) => {
            // Get modal elements once the DOM is ready
            modal = document.getElementById('modal');
            modalBackdrop = document.getElementById('modal-backdrop');
            modalTitle = document.getElementById('modal-title');
            modalBody = document.getElementById('modal-body');

             // Add event listener for Escape key only after DOM is loaded
            document.addEventListener('keydown', function(event) {
                // Check if modal exists and is not hidden
                if (event.key === "Escape" && modal && !modal.classList.contains('hidden')) {
                    closeModal();
                }
            });
        });


        /**
         * Opens the modal and populates it with details for the given phase ID.
         * @param {number} phaseId - The ID of the phase (1-6).
         */
        function openModal(phaseId) {
            // Ensure modal elements have been found (DOM is ready)
            if (!modal || !modalBackdrop || !modalTitle || !modalBody) {
                console.error("Modal elements not found. DOM might not be ready yet.");
                return;
            }

            const data = phaseDetails[phaseId];
            if (data) {
                modalTitle.textContent = data.title;
                modalBody.innerHTML = data.details; // Use innerHTML because details contain HTML tags
                // Use visible/hidden classes for opacity transition
                modal.classList.remove('hidden');
                modalBackdrop.classList.remove('hidden');
                // Force reflow to ensure transition plays
                void modal.offsetWidth;
                modal.classList.add('visible');
                modalBackdrop.classList.add('visible');
                // Scroll modal body to top
                modalBody.scrollTop = 0;


            } else {
                console.error("Phase details not found for ID:", phaseId);
            }
        }

        /**
         * Closes the modal.

             // Ensure modal elements have been found
